  // A named window is the only practical best-effort reuse mechanism available
  // from a userscript. Violentmonkey's GM_openInTab does not focus existing tabs.
  function openPCMS() {
    if (IS_APP) { window.focus(); return; }
    try {
      // An empty URL reuses the named tab without reloading an active UI.
      const target = window.open('', TAB_NAME);
      if (target && !target.closed) {
        try {
          if (target.location.href !== APP_URL) target.location.replace(APP_URL);
          target.focus();
          return;
        } catch (_) {
          // A named tab has navigated cross-origin: do not hijack it.
        }
      }
    } catch (_) {}
    try { GM_openInTab(APP_URL, { active: true, insert: true }); }
    catch (error) { console.error('[PCMS UI demo] Could not open tab', error); }

    const r=g.blockReason || (g.workflow==='blocked'?'unclassified_hold':g.sync==='conflict'?'source_conflict':null);

  const CASE_ORDER = Object.freeze({identity:0,unlisted:1,uncertain:2,conflict:3,blocked:4,review:5});
  function attention() {
    if (demo.demoScenario==='loading') return null; // state not yet verified
    if (demo.demoScenario==='quiet') return [];
    const cases=[];
    if (demo.demoScenario==='signed-out') cases.push({key:'account:login',type:'identity',title:'Sign in required',detail:'Perchance operations paused',button:'Sign in',action:'sign-in',scope:'account'});
    if (demo.demoScenario==='unconfirmed') cases.push({key:'account:binding',type:'identity',title:'Confirm account',detail:'Owner operations paused',button:'Confirm',action:'session',scope:'account'});
    if (demo.demoScenario==='wrong-account') cases.push({key:'account:mismatch',type:'identity',title:'Wrong account',detail:'Account binding needs verification',button:'Review',action:'settings',scope:'account'});
    for(const g of demo.generators) {
      const issue=blockCase(g);
      if(issue)cases.push(issue);
      if(g.sync==='conflict'&&g.blockReason!=='source_conflict')cases.push({key:`generator:${g.id}:source_conflict`,type:'conflict',reason:'source_conflict',id:g.id,title:'Source conflict',detail:g.slug,button:'Compare revisions',scope:'generator'});
      if(g.workflow==='review')cases.push({key:`generator:${g.id}:review`,type:'review',id:g.id,title:'Review required',detail:g.slug,button:'Review',scope:'generator'});
    }
    const byTarget=new Map();
    for(const item of cases){
      const target=item.scope==='account'?item.key:`generator:${item.id}`;
      const previous=byTarget.get(target);
      if(!previous) byTarget.set(target,{...item,related:[]});
      else if(CASE_ORDER[item.type]<CASE_ORDER[previous.type]) byTarget.set(target,{...item,related:[previous,...(previous.related||[])]});
      else previous.related.push(item);
    }
    return [...byTarget.values()].sort((a,b)=>CASE_ORDER[a.type]-CASE_ORDER[b.type] || (a.id??0)-(b.id??0) || a.key.localeCompare(b.key));
  }

  function performImport() {
    const gate=importGate();if(gate){toast(gate);return;}
    const candidates=demo.importCandidates||[];
    const selected=candidates.filter(c=>demo.importSelected.has(c.id));
    if(!selected.length){toast('Select at least one eligible generator.');return;}
    const results={};
    for(const c of candidates){
      if(!demo.importSelected.has(c.id)){results[c.id]={state:'skipped',reason:'Not selected'};continue;}
      if(!importEligibility(c).ready){results[c.id]={state:'skipped',reason:'Preflight changed; import blocked.'};continue;}
      // A failed acknowledgement is NOT proof that a remote write failed.
      // It must not be retried until a read reconciles the target folder.
      if(c.id===109){c.check='verify';results[c.id]={state:'uncertain',reason:'Commit result unknown; inspect GitHub main before retry.'};continue;}
      c.check='managed';demo.generators.push({id:c.id,slug:c.slug,listed:c.listed,excluded:false,workflow:'done',sync:'synced',hold:'',last:'Imported (demo)',attempts:0});
      results[c.id]={state:'imported',reason:c.listed?'Listed · Refresher eligible':'Unlisted · excluded from Refresher'};
    }
    demo.importResults=results;demo.importPhase='results';demo.importFilter='processed';demo.importSearch='';render();
    mount.querySelector('.main')?.scrollTo(0,0);
  }
  function recheckImport(id) {

    // Demo simulation of an authoritative negative folder check, not a live check.
    c.check='ready';demo.importResults[id]={state:'retryable',reason:'Demo read confirms folder absent; explicit retry permitted.'};render();
  }
  function retryImport(id) {
    const c=(demo.importCandidates||[]).find(x=>x.id===id);const r=demo.importResults[id];
    if(!c||r?.state!=='retryable'||importGate()||!importEligibility(c).ready)return;

    demo.importResults[id]={state:'imported',reason:c.listed?'Listed · Refresher eligible':'Unlisted · excluded from Refresher'};
    render();
  }

  // Refresher v1.5 UI model. All timestamps/operations are simulated, never authoritative.
  // The queue retains order across views and applies membership changes incrementally.
  function refreshQueue() {
    const eligible=active(), known=new Set(eligible.map(g=>g.id));
    const ordered=(demo.refQueue||[]).filter(id=>known.has(id));
    const present=new Set(ordered);
    for(const g of eligible)if(!present.has(g.id)){ordered.push(g.id);present.add(g.id);}
    demo.refQueue=ordered;
    return ordered.map(genById).filter(Boolean);
  }
  function rollingAttempts(id=null, time=demo.refClock) {
    return demo.refHistory.filter(a=>a.at>time-3600000 && a.at<=time && (id==null || a.id===id));
  }
  function refresherSnapshot() {
    const members=refreshQueue(), n=members.length, cap=demo.cap;
    const used=rollingAttempts().length;
    const perGen=n?Math.ceil(cap/n):null;
    const rate=n?cap/n:null;
    const interval=n?n*3600000/cap:null;
    const op=operationSnapshot().items[0];
    const verified=op.status==='Running' && ownerSessionVerified() && !demo.paused;
    const last=demo.refHistory[0]?.at;
    const paced=last==null?demo.refClock:last+3600000/cap;
    let accountDue=Math.max(demo.refClock,paced);
    const accountWindow=rollingAttempts();
    if(accountWindow.length>=cap){
      const expiry=accountWindow.map(a=>a.at).sort((a,b)=>a-b)[accountWindow.length-cap];
      accountDue=Math.max(accountDue,expiry+3600000+1);
    }
    // Earliest time at which at least one current queue member fits its rolling-hour cap.
    let memberDue=null;
    if(n){
      memberDue=Math.min(...members.map(g=>{
        const h=rollingAttempts(g.id);
        return h.length<perGen?demo.refClock:h.map(a=>a.at).sort((a,b)=>a-b)[h.length-perGen]+3600000+1;
      }));
    }
    const nextAt=verified&&n?Math.max(accountDue,memberDue):null;
    return {members,n,cap,used,perGen,rate,interval,op,nextAt,verified};
  }
  function formatRefTime(timestamp) {
    return new Date(timestamp).toLocaleTimeString('en-GB',{hour:'2-digit',minute:'2-digit'});
  }
  function formatWait(ms) {
    if(ms<=0)return 'Ready';
    const minutes=Math.ceil(ms/60000);
    if(minutes<60)return `In ${minutes} min`;
    const hours=Math.floor(minutes/60), remainder=minutes%60;
    return remainder?`In ${hours}h ${remainder}m`:`In ${hours}h`;
  }
  function refreshIneligibleReason(g) {
    if(g.excluded)return ['Excluded','neutral'];
    if(!g.listed)return ['Unlisted','neutral'];
    if(g.blockReason==='save_uncertain')return ['Uncertain Save','red'];

  function simulateRefresherSlot() {
    const snap=refresherSnapshot();
    if(!snap.verified||!snap.n)return {message:'Refresher cannot execute a simulated slot in this state.'};
    demo.refClock=Math.max(demo.refClock,snap.nextAt??demo.refClock);
    const queue=refreshQueue(), perGen=Math.ceil(demo.cap/queue.length);
    if(rollingAttempts().length>=demo.cap)return {message:'Rolling-hour cap reached. Nothing was attempted.'};
    let candidate=null;
    // Inspect at most one FIFO pass: capped members rotate to tail in the same slot.
    for(let i=0;i<queue.length;i++){
      const id=demo.refQueue.shift(),g=genById(id);
      demo.refQueue.push(id);
      if(g&&rollingAttempts(g.id).length<perGen){candidate=g;break;}
    }
    if(!candidate)return {message:'No generator is eligible for this slot. Nothing was attempted.'};
    const outcome=demo.demoScenario==='refresher-failed'?'Failed':demo.demoScenario==='refresher-uncertain'?'Unknown':'Saved';
    const reason=outcome==='Failed'?'Native Save rejected':outcome==='Unknown'?'Native Save acknowledgement missing':'';
    demo.refHistory.unshift({id:candidate.id,at:demo.refClock,outcome,reason});
    if(outcome==='Unknown'){
      candidate.blockReason='save_uncertain';candidate.hold='Uncertain Save';
    }
    appendJournal('Refresher',outcome,candidate.slug,formatRefTime(demo.refClock));
    refreshQueue();
    return {message:`Demo ${outcome.toLowerCase()} · ${candidate.slug}. ${outcome==='Failed'?'Attempt counted; no immediate retry.':outcome==='Unknown'?'Generator held until verified.':'Moved to end of FIFO.'}`};
  }

      return 'https://github.com/'+segments[0]+'/'+segments[1].replace(/\.git$/,'');
    }catch(_){return null;}
  }

    const scenario=demo.demoScenario;

    const items=demo.generators.filter(g=>['pending','conflict'].includes(g.sync)).map(g=>{
      const conflict=g.sync==='conflict';
      const uncertain=scenario==='sync-write-uncertain'&&g.id===9&&demo.syncEvidence[g.id]!=='absent';
      const state=unverified?'unknown':uncertain?'uncertain':demo.syncDisposition[g.id]==='remote-queued'?'publishing':conflict?'conflict':g.blockReason==='repository_change'?'target_held':offline?'offline':'pending';
      const originalRepo=g.repositoryBound||'https://github.com/example/pcms-generators';
      return {id:g.id,slug:g.slug,listed:g.listed,kind:'generator',state,repo:originalRepo,
        baseline:conflict?'4e91c2a':'3b6e9f1',local:conflict?'local · a7e8d21':'local · b412c66',remote:conflict?'main · 7c2de90':'main · 3b6e9f1',
        localReady:g.listed&&(g.workflow==='done'||demo.syncDisposition[g.id]==='remote-queued')&&(!g.hold||g.blockReason==='source_conflict'||demo.syncDisposition[g.id]==='remote-queued'),absentProven:demo.syncEvidence[g.id]==='absent',
        reason:conflict?'Source changed on GitHub since local approval':uncertain?'GitHub commit acknowledgement missing':offline?'Connection unavailable':'Commit awaiting remote verification'};
    });
    if(scenario==='sync-storage-failed'){
      items.push({id:5,slug:'dreamscape-xl',listed:true,kind:'storage',state:'storage_failed',repo:demo.settings.repo,
        baseline:'not recorded',local:'draft not retained',remote:'not checked',localReady:false,reason:'Full source snapshot could not be persisted; native Save must not start'});
    }
    for(const item of demo.syncCompleted)items.push({...item});
    return items.sort((a,b)=>({conflict:0,uncertain:1,storage_failed:2,offline:3,pending:4,target_held:4,unknown:5,synced:6}[a.state]??9)-({conflict:0,uncertain:1,storage_failed:2,offline:3,pending:4,target_held:4,unknown:5,synced:6}[b.state]??9)||a.id-b.id);
  }

      if(['ArrowDown','ArrowUp','Enter',' '].includes(key)){
        e.preventDefault();const pickerKey=target.dataset.pickerTrigger;
        if(demo.pickerOpen!==pickerKey){demo.pickerOpen=pickerKey;render();}
        placePicker(pickerKey);
        const selected=mount.querySelector(`[data-picker-option="${pickerKey}"][aria-selected="true"]`);
        selected?.focus({preventScroll:true});
        if(key==='ArrowUp')movePickerFocus(pickerKey,-1);
        if(key==='ArrowDown')movePickerFocus(pickerKey,1);
        return;
      }
    }
    if(target?.dataset?.pickerOption){
      const pkey=target.dataset.pickerOption;
      if(key==='ArrowDown'||key==='ArrowUp'){e.preventDefault();movePickerFocus(pkey,key==='ArrowDown'?1:-1);return;}
      if(key.length===1&&!e.ctrlKey&&!e.altKey&&!e.metaKey&&/[\p{L}\p{N}]/u.test(key)){
        const options=[...mount.querySelectorAll(`[data-picker-option="${pkey}"]`)];
        const first=options.find(x=>x.textContent.trim().toLocaleLowerCase().startsWith(key.toLocaleLowerCase()));
        if(first){e.preventDefault();options.forEach(x=>x.tabIndex=x===first?0:-1);first.focus({preventScroll:true});first.scrollIntoView({block:'nearest'});return;}
      }
      if(key==='Home'||key==='End'){e.preventDefault();const opts=[...mount.querySelectorAll(`[data-picker-option="${pkey}"]`)];opts.forEach((o,i)=>o.tabIndex=i===(key==='Home'?0:opts.length-1)?0:-1);opts[key==='Home'?0:opts.length-1]?.focus();return;}
      if(key==='Escape'){e.preventDefault();e.stopPropagation();closePicker({focus:true});return;}
      if(key==='Tab'){closePicker({focus:true});return;}
    }
    if(demo.pickerOpen&&key==='Escape'){e.preventDefault();closePicker({focus:true});return;}
    const modal = mount.querySelector('.dialog') || mount.querySelector('.drawer');
    if (modal && e.key === 'Tab') {
      const targets = Array.from(modal.querySelectorAll('button:not(:disabled),input:not(:disabled),select:not(:disabled)'));
      if (targets.length) {
        const first=targets[0],last=targets[targets.length-1];
        const current=shadow.activeElement;
        if (e.shiftKey && (current===first || !modal.contains(current))) {e.preventDefault();last.focus();return;}
        if (!e.shiftKey && (current===last || !modal.contains(current))) {e.preventDefault();first.focus();return;}
      }
    }
    if(e.key==='Escape'&&(demo.dialog||demo.panel||demo.issuePanel||demo.operationPanel||demo.eventPanel||demo.journalDetail)){
      e.preventDefault();e.stopPropagation();
      if(demo.dialog)demo.dialog=null;
      else if(demo.issuePanel)demo.issuePanel=null;
      else if(demo.operationPanel)demo.operationPanel=null;
      else if(demo.journalDetail||demo.eventPanel){demo.journalDetail=null;demo.eventPanel=null;}
      else demo.panel=null;
      demo.managementError='';render();return;
    }
    if(e.key==='/'&&!e.altKey&&!e.ctrlKey&&!e.metaKey&&!e.isComposing&&
      !modal&&!demo.pickerOpen&&!['INPUT','TEXTAREA'].includes(shadow.activeElement?.tagName)&&
      !shadow.activeElement?.isContentEditable){
      if(demo.tab!=='generators')navigate('generators');
      e.preventDefault();mount.querySelector('#generator-search')?.focus();
    }
  });
  render();
  console.info('[PCMS UI demo] Mounted dedicated interface prototype', {version:VERSION, url:APP_URL});
  }

  // Boot as soon as the browser creates <body>. No dependency on a completed
  // Perchance page load, and never mutate a partially parsed document tree.
  let booted = false;
  const boot = () => {
    if (booted || !document.body) return;
    booted = true;
    try { startApp(); }
    catch (error) {
      console.error('[PCMS UI demo] UI mount failed', error);
      const body = document.body;
      if (body) {
        const failure = document.createElement('section');
        failure.id = 'pcms-ui-mockup-failure';
        failure.style.cssText = 'position:fixed;inset:0;z-index:2147483647;background:#fff;color:#202631;padding:40px;font:14px/1.5 system-ui';
        const heading = document.createElement('h1'); heading.textContent = 'PCMS mockup failed to initialize';
        const details = document.createElement('pre'); details.textContent = String(error?.stack || error);
        details.style.cssText = 'white-space:pre-wrap;overflow-wrap:anywhere';
        failure.append(heading,details);
        body.append(failure);
      }
    }
  };
  if (document.body) boot();
  else {
    const observer = new MutationObserver(() => {
      if (document.body) { observer.disconnect(); boot(); }
    });
    observer.observe(document, { childList: true, subtree: true });
    document.addEventListener('DOMContentLoaded', () => { observer.disconnect(); boot(); }, { once: true });
  }
})();
